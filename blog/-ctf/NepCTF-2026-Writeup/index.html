下发一个神秘 novel.txt，内容是一篇网文，偶然发现使用记事本打开和使用 vim 打开内容不太一样

估计是 unicode 藏东西了，恰好遇到兼容性差的 notepad，于是显现出来了

这种情况直接把前几个自然段丢给 LLMs，因为 LLMs 对这些 unicode 字符敏感。很快发现用 unicode 变体选择符藏东西了，尝试将 \uFE00-\uFE0F 映射为 0x0-0xF，解码为 Hex

print(bytes.fromhex(hex_str).decode('utf-8', errors='replace'))

下发一张 jpg 图片和 sketch_may27b_send.ino.merged.bin，前者搜了一下发现是 ESP32-S3 开发板，此外图片的 comment 里藏了 ch18 id19 的信息，EOF 后面藏了 ch19 id18 的信息，感觉是矛盾的

app 数据大致为一个 Godox 无线引闪发射器，使用 SI24R1 (nRF24L01+) 2.4GHz 的无线模块。发射模式下可以通过 raw <16 hex...> 发送原始的数据包，并且给出了一个 example

import asyncio
import aiohttp
from bs4 import BeautifulSoup

BASE_URL = "https://www.yxdown.com/SoftView/SoftView_{}.html"
TARGET = "炒地皮"
START, END, CONCURRENT = 1, 100000, 20
found = False

async def fetch(session, num, sem):
    global found
    if found:
        return

    url = BASE_URL.format(num)
    async with sem:
        try:
            async with session.get(url, timeout=10, headers={"User-Agent": "Mozilla/5.0"}) as r:
                if r.status != 200:
                    return

                soup = BeautifulSoup(await r.text(errors="ignore"), "html.parser")
                h1_text = next((h.get_text(strip=True) for h in soup.find_all('h1') if TARGET in h.get_text(strip=True)), None)

                if h1_text:
                    found = True
                    print(f"\n✅ 找到: {url}\n标题: {h1_text}\n编号: {num}")
                    with open(f"found_{num}.html", "w", encoding="utf-8") as f:
                        f.write(await r.text())
                    with open("found_number.txt", "w") as f:
                        f.write(str(num))
        except:
            pass

async def main():
    sem = asyncio.Semaphore(CONCURRENT)
    async with aiohttp.ClientSession(connector=aiohttp.TCPConnector(limit=CONCURRENT)) as session:
        tasks = []
        for i in range(START, END + 1):
            if found:
                break
            tasks.append(fetch(session, i, sem))
            if len(tasks) >= CONCURRENT * 5:
                await asyncio.gather(*tasks)
                tasks.clear()
                if found:
                    break
        if tasks and not found:
            await asyncio.gather(*tasks)

    if not found:
        print(f"\n❌ 未找到: {TARGET}")

if __name__ == "__main__":
    print(f"🔍 扫描 {TARGET} ({START}-{END})")
    asyncio.run(main())
