    import socket
import re

host = "10.0.2.8"
port = 1337

s = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
s.connect((host, port))

pattern = re.compile(r'\((\d+), \'([+\-*/])\', (\d+)\)')

count = 0
buffer = ""
# 其实有 1001 次提问
while count <= 1001:
    data = s.recv(1024).decode('utf-8')
    buffer += data
    print(data, end="")

    if ">" in buffer:
        match = pattern.search(buffer)
        if match:
            num1 = int(match.group(1))
            operator = match.group(2)
            num2 = int(match.group(3))

            result = None
            if operator == '+':
                result = num1 + num2
            elif operator == '-':
                result = num1 - num2
            elif operator == '*':
                result = num1 * num2
            elif operator == '/':
                result = num1 // num2

            s.send(f"{result}\n".encode('utf-8'))
            buffer = ""
        count += 1

while True:
    data = s.recv(1024).decode('utf-8')
    if not data:
        break
    print(data, end="")

s.close()
